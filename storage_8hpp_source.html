216 [[nodiscard]] auto &element_at(const stl::size_t pos) const {
217 return payload[pos / traits_type::page_size][fast_mod(pos, traits_type::page_size)];
218 }

220 auto assure_at_least(const stl::size_t pos) {
221 const auto idx = pos / traits_type::page_size;
222
223 if(!(idx < payload.size())) {
224 auto curr = payload.size();
225 allocator_type allocator{get_allocator()};
226 payload.resize(idx + 1u, nullptr);
227
228 ENTT_TRY {
229 for(const auto last = payload.size(); curr < last; ++curr) {
230 payload[curr] = alloc_traits::allocate(allocator, traits_type::page_size);
231 }
232 }
233 ENTT_CATCH {
234 payload.resize(curr);
235 ENTT_THROW;
236 }
237 }
238
239 return payload[idx] + fast_mod(pos, traits_type::page_size);
240 }

242 template<typename... Args>
243 auto emplace_element(const Entity entt, const bool force_back, Args &&...args) {
244 const auto it = base_type::try_emplace(entt, force_back);
245
246 ENTT_TRY {
247 auto *elem = stl::to_address(assure_at_least(static_cast<size_type>(it.index())));
248 entt::uninitialized_construct_using_allocator(elem, get_allocator(), stl::forward<Args>(args)...);
249 }
250 ENTT_CATCH {
251 base_type::pop(it, it + 1u);
252 ENTT_THROW;
253 }
254
255 return it;
256 }

258 void shrink_to_size(const stl::size_t sz) {
259 const auto from = (sz + traits_type::page_size - 1u) / traits_type::page_size;
260 allocator_type allocator{get_allocator()};
261
262 if constexpr(!stl::is_trivially_destructible_v<element_type>) {
263 for(auto pos = sz, length = base_type::size(); pos < length; ++pos) {
264 if constexpr(traits_type::in_place_delete) {
265 if(base_type::data()[pos] != tombstone) {
266 alloc_traits::destroy(allocator, stl::addressof(element_at(pos)));
267 }
268 } else {
269 alloc_traits::destroy(allocator, stl::addressof(element_at(pos)));
270 }
271 }
272 }
273
274 for(auto pos = from, last = payload.size(); pos < last; ++pos) {
275 alloc_traits::deallocate(allocator, payload[pos], traits_type::page_size);
276 }
277
278 payload.resize(from);
279 payload.shrink_to_fit();
280 }

287 void move_to(const stl::size_t lhs, const stl::size_t rhs) {
288 auto &elem = element_at(lhs);
289 allocator_type allocator{get_allocator()};
290 entt::uninitialized_construct_using_allocator(stl::to_address(assure_at_least(rhs)), allocator, stl::move(elem));

299 void swap_or_move([[maybe_unused]] const stl::size_t from, [[maybe_unused]] const stl::size_t to) override {
300 static constexpr bool is_pinned_type = !(stl::is_move_constructible_v<Type> && stl::is_move_assignable_v<Type>);
301 // use a runtime value to avoid compile-time suppression that drives the code coverage tool crazy
302 ENTT_ASSERT((from + 1u) && !is_pinned_type, "Pinned type");
303
304 if constexpr(!is_pinned_type) {
305 if constexpr(traits_type::in_place_delete) {
306 (base_type::operator[](to) == tombstone) ? move_to(from, to) : swap_at(from, to);
307 } else {
308 swap_at(from, to);
309 }
310 }

319 void pop(underlying_iterator first, underlying_iterator last) override {
320 for(allocator_type allocator{get_allocator()}; first != last; ++first) {
321 // cannot use first.index() because it would break with cross iterators
322 auto &elem = element_at(base_type::index(*first));
323
324 if constexpr(traits_type::in_place_delete) {
325 base_type::in_place_pop(*first);
326 alloc_traits::destroy(allocator, stl::addressof(elem));
327 } else if constexpr(stl::is_trivially_destructible_v<element_type>) {
328 elem = stl::move(element_at(base_type::size() - 1u));
329 base_type::swap_and_pop(*first);
330 } else {
331 auto &other = element_at(base_type::size() - 1u);
332 // destroying on exit allows reentrant destructors
333 [[maybe_unused]] auto unused = stl::exchange(elem, stl::move(other));
334 alloc_traits::destroy(allocator, stl::addressof(other));
335 base_type::swap_and_pop(*first);
336 }
337 }

341 void pop_all() override {
342 if constexpr(stl::is_trivially_destructible_v<element_type>) {
343 base_type::pop_all();
344 } else {
345 allocator_type allocator{get_allocator()};
346
347 for(auto first = base_type::begin(); !(first.index() < 0); ++first) {
348 if constexpr(traits_type::in_place_delete) {
349 if(*first != tombstone) {

352 }

356 }
