757template<typename Type>
758[[nodiscard]] ENTT_CONSTEVAL bool dispatch_is_equality_comparable() {
759 // NOLINTBEGIN(modernize-use-transparent-functors)
760 if constexpr(stl::is_array_v<Type>) {
761 return false;
762 } else if constexpr(is_complete_v<stl::tuple_size<stl::remove_const_t<Type>>>) {
763 if constexpr(has_tuple_size_value<Type>::value) {
764 return maybe_equality_comparable<Type>(0) && unpack_maybe_equality_comparable<Type>(stl::make_index_sequence<stl::tuple_size<Type>::value>{});
765 } else {
766 return maybe_equality_comparable<Type>(0);
767 }
768 } else if constexpr(has_value_type<Type>::value) {
769 if constexpr(is_iterator_v<Type> || stl::is_same_v<typename Type::value_type, Type> || dispatch_is_equality_comparable<typename Type::value_type>()) {
770 return maybe_equality_comparable<Type>(0);
771 } else {
772 return false;
773 }
774 } else {
775 return maybe_equality_comparable<Type>(0);
776 }
777 // NOLINTEND(modernize-use-transparent-functors)
778}
